cmake_minimum_required(VERSION 3.25)
project(ConsoleLAN VERSION 1.2.1 LANGUAGES CXX)
# CMake can misdecode the UTF-8 /showIncludes prefix with Spanish-only MSVC.
# Keep Ninja's dependency parser consistent with the compiler's actual bytes.
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "^Nota:")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Nota: inclusión del archivo: ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets)
qt_standard_project_setup()
set(CMAKE_AUTORCC ON)
add_library(consolelan_core src/protocol.cpp src/ethernet.cpp src/room.cpp src/store.cpp src/access_policy.cpp)
target_include_directories(consolelan_core PUBLIC src)
target_link_libraries(consolelan_core PUBLIC Qt6::Core)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
endif()
if(WIN32)
  set(NPCAP_SDK "${CMAKE_SOURCE_DIR}/deps/npcap-sdk-1.17" CACHE PATH "Npcap SDK root")
  find_path(PCAP_INCLUDE pcap.h PATHS "${NPCAP_SDK}/Include" REQUIRED NO_DEFAULT_PATH)
  add_library(consolelan_platform src/network.cpp src/capture.cpp src/worker.cpp src/access_controller.cpp)
  target_include_directories(consolelan_platform PRIVATE "${PCAP_INCLUDE}")
  target_compile_definitions(consolelan_platform PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(consolelan_platform PUBLIC consolelan_core ws2_32 iphlpapi shell32 advapi32)
  add_library(consolelan_access src/access_native.cpp)
  target_include_directories(consolelan_access PRIVATE "${CMAKE_SOURCE_DIR}/deps/wintun-0.14.1/wintun/include")
  target_link_libraries(consolelan_access PUBLIC consolelan_platform)
  add_executable(ConsoleLAN.Access WIN32 src/access_main.cpp)
  target_link_libraries(ConsoleLAN.Access PRIVATE consolelan_access)
  set_target_properties(ConsoleLAN.Access PROPERTIES WIN32_EXECUTABLE FALSE)
  add_custom_command(TARGET ConsoleLAN.Access POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_SOURCE_DIR}/deps/wintun-0.14.1/wintun/bin/amd64/wintun.dll" "$<TARGET_FILE_DIR:ConsoleLAN.Access>/wintun.dll")
  add_library(consolelan_ui src/window.cpp src/presentation.cpp src/presentation.qrc)
  target_link_libraries(consolelan_ui PUBLIC consolelan_platform Qt6::Widgets dwmapi)
  target_compile_definitions(consolelan_ui PUBLIC CONSOLELAN_VERSION="${PROJECT_VERSION}")
  qt_add_executable(ConsoleLAN WIN32 src/main.cpp)
  target_link_libraries(ConsoleLAN PRIVATE consolelan_ui)
  set_target_properties(ConsoleLAN PROPERTIES WIN32_EXECUTABLE TRUE)
  install(TARGETS ConsoleLAN RUNTIME DESTINATION .)
  qt_generate_deploy_app_script(TARGET ConsoleLAN OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
  add_library(consolelan_access_test src/access_native.cpp)
  target_include_directories(consolelan_access_test PRIVATE "${CMAKE_SOURCE_DIR}/deps/wintun-0.14.1/wintun/include")
  target_compile_definitions(consolelan_access_test PRIVATE CONSOLELAN_NATIVE_TESTS)
  target_link_libraries(consolelan_access_test PUBLIC consolelan_platform)
  add_executable(access_native_tests tests/access_native_tests.cpp)
  target_link_libraries(access_native_tests PRIVATE consolelan_access_test)
  add_executable(network_tests tests/network_tests.cpp)
  target_link_libraries(network_tests PRIVATE consolelan_platform)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE consolelan_ui Qt6::Test)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE consolelan_core)
add_test(NAME core COMMAND core_tests)
add_executable(access_tests tests/access_tests.cpp)
target_link_libraries(access_tests PRIVATE consolelan_core)
add_test(NAME local_access_policy COMMAND access_tests)
if(WIN32)
  add_test(NAME winsock_loopback COMMAND network_tests)
  add_test(NAME local_access_native COMMAND access_native_tests)
  set_tests_properties(local_access_native PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME user_interface COMMAND ui_tests -o ${CMAKE_CURRENT_BINARY_DIR}/ui-tests.xml,junitxml -o -,txt)
endif()
